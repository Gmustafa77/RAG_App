# ==============================================================================
# requirements.txt
# ------------------------------------------------------------------------------
# This file lists every third-party Python package the Aetna Insurance Virtual
# Assistant needs to run. Install them all with:
#
#       pip install -r requirements.txt
#
# Why pin versions?
#   RAG apps chain together many libraries (LangChain, ChromaDB, OpenAI SDK,
#   Streamlit, PyMuPDF...). These libraries evolve quickly and sometimes make
#   breaking changes between versions. Pinning versions means the app you build
#   today will still work the same way next month.
# ==============================================================================

# --- Frontend --------------------------------------------------------------
streamlit           # Web UI framework used to build the chat interface

# --- LLM + Embeddings (OpenAI) ----------------------------------------------
openai              # Official OpenAI Python SDK (GPT-4o + embeddings)

# --- Orchestration Framework -------------------------------------------------
langchain            # Core LangChain framework (chains, prompts, memory)
langchain-openai       # LangChain <-> OpenAI integration (chat + embeddings)
langchain-google-genai # LangChain <-> Google Gemini integration (free embeddings)
langchain-community    # Community integrations (used for some loaders/utilities)
langchain-chroma       # LangChain <-> ChromaDB integration

# --- Vector Database ---------------------------------------------------------
chromadb               # Local, on-disk vector database for storing embeddings

# --- PDF Parsing ---------------------------------------------------------------
PyMuPDF             # Fast PDF text/metadata extraction (imported as "fitz")

# --- Utilities -----------------------------------------------------------------
python-dotenv           # Loads environment variables (e.g., API keys) from .env
tiktoken               # Token counting used for chunking & cost estimation
pydantic                # Data validation, used internally by LangChain
tenacity               # Retry logic for flaky network/API calls
